export function normalizePath(path) {
  const raw = String(path || '/')
    .split('#')[0]
    .split('?')[0]
    .trim()

  if (!raw || raw === '/') return '/'

  const prefixed = raw.startsWith('/') ? raw : \`/${raw}\`
  return prefixed.replace(/\/+$/, '') || '/'
}

export function stripLocalePrefix(path) {

  const basePath = stripLocalePrefix(path)

}
